create table if not exists public.students (
  id text primary key,
  name text not null,
  whatsapp text unique,
  pin text not null,
  city text,
  access_status text default 'free',
  completed_classes jsonb default '[1]'::jsonb,
  current_lesson int default 1,
  cp1_status text default 'not_started',
  cp2_status text default 'not_started',
  notes text,
  joined_at timestamptz default now(),
  last_active timestamptz default now()
);
alter table public.students enable row level security;
create policy "Allow public read/write" on public.students for all using (true) with check (true);
